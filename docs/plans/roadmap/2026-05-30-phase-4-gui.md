# Phase 4: GUI / Electron — Implementation Plan

> 完成状态以 [CHANGELOG](../../CHANGELOG.md) 为准。


**Goal:** Build a working Electron desktop app with progressive three-layer disclosure (Surface → Context → Control), secure IPC isolation, and kernel integration.

**Architecture:** Electron main process hosts the Agent Kernel and exposes a whitelisted IPC API. The preload script bridges via `contextBridge`. The renderer uses vanilla HTML/CSS/JS (no build step) with a UI State Adapter that transforms kernel events into view models for each layer. Surface Layer is the default chat view; Context Layer auto-expands when the agent is working; Control Layer opens for plan/permission approval.

**Tech Stack:** Electron 30+, Node.js >= 20, vanilla HTML/CSS/JS renderer, Phase 0-3 kernel modules.

```js
// gui/main.js — Electron main process
const { app, BrowserWindow, ipcMain, dialog } = require("electron");
const path = require("path");

// Dynamic import for ES module kernel
let kernel = null;
let projectRoot = null;

async function initKernel() {
  // Use the project root from the first argument or CWD
  projectRoot = process.argv.find(a => a.startsWith("--project="))
    ?.split("=")[1] || process.cwd();

  // Dynamic import of the ES module kernel
  const { createKernel } = await import(
    path.join(__dirname, "..", "src", "kernel", "kernel-api.js")
  );
  kernel = await createKernel(projectRoot, { config: { allowMissingKey: true } });
  return kernel;
}

async function createWindow() {
  const win = new BrowserWindow({
    width: 900,
    height: 700,
    minWidth: 400,
    minHeight: 400,
    webPreferences: {
      preload: path.join(__dirname, "preload.js"),
      nodeIntegration: false,
      contextIsolation: true,
      sandbox: true
    },
    title: "DeepSeek Code",
    frame: true
  });

  // Initialize kernel
  try {
    await initKernel();
    console.log("Kernel initialized for:", projectRoot);
  } catch (err) {
    console.error("Kernel init failed:", err.message);
  }

  // Register IPC handlers — whitelisted API
  registerIpcHandlers();

  win.loadFile(path.join(__dirname, "renderer", "index.html"));
  return win;
}

function registerIpcHandlers() {
  // agent.send
  ipcMain.handle("agent:send", async (_event, message, opts) => {
    if (!kernel) return { error: "Kernel not ready" };
    try {
      const result = await kernel.agent.send(message, opts);
      return { result };
    } catch (err) {
      return { error: err.message };
    }
  });

  // agent.approve
  ipcMain.handle("agent:approve", (_event, id, decision) => {
    if (!kernel) return { error: "Kernel not ready" };
    kernel.agent.approve(id, decision);
    return { ok: true };
  });

  // agent.interrupt
  ipcMain.handle("agent:interrupt", () => {
    if (!kernel) return { error: "Kernel not ready" };
    kernel.agent.interrupt();
    return { ok: true };
  });

  // session.getTimeline
  ipcMain.handle("session:timeline", async (_event, count) => {
    if (!kernel) return [];
    return kernel.session.getTimeline(count || 20);
  });

  // context.getSnapshot
  ipcMain.handle("context:snapshot", async () => {
    if (!kernel) return { units: [] };
    return kernel.context.getSnapshot();
  });

  // model.getUsageStats
  ipcMain.handle("model:usage", () => {
    if (!kernel?.modelProvider) return {};
    return kernel.modelProvider.getUsageStats();
  });

  // config.get
  ipcMain.handle("config:get", () => {
    if (!kernel) return {};
    const c = kernel.config;
    return {
      model: c.model,
      baseUrl: c.baseUrl,
      thinking: c.thinking?.type || "disabled",
      reasoningEffort: c.reasoningEffort
    };
  });

  // orch.getState
  ipcMain.handle("orchestrator:state", () => {
    if (!kernel?.orchestrator) return { current: "idle", autonomy: "gated", channel: null };
    return kernel.orchestrator.getState();
  });

  // session.subscribe — push events to renderer
  if (kernel) {
    kernel.session.subscribe((event) => {
      const win = BrowserWindow.getAllWindows()[0];
      if (win && !win.isDestroyed()) {
        win.webContents.send("kernel:event", event);
      }
    });
  }
}

app.whenReady().then(createWindow);
app.on("window-all-closed", () => app.quit());
```

#### Step 3: Create `gui/preload.js`

```js
// gui/preload.js — Secure context bridge
const { contextBridge, ipcRenderer } = require("electron");

contextBridge.exposeInMainWorld("deepseek", {
  // Agent
  send: (message, opts) => ipcRenderer.invoke("agent:send", message, opts),
  approve: (id, decision) => ipcRenderer.invoke("agent:approve", id, decision),
  interrupt: () => ipcRenderer.invoke("agent:interrupt"),

  // Session
  getTimeline: (count) => ipcRenderer.invoke("session:timeline", count),
  onKernelEvent: (callback) => {
    const handler = (_event, data) => callback(data);
    ipcRenderer.on("kernel:event", handler);
    return () => ipcRenderer.removeListener("kernel:event", handler);
  },

  // Context
  getSnapshot: () => ipcRenderer.invoke("context:snapshot"),
  getUsage: () => ipcRenderer.invoke("model:usage"),
  getConfig: () => ipcRenderer.invoke("config:get"),
  getState: () => ipcRenderer.invoke("orchestrator:state"),
});
```

#### Step 2: Create `gui/renderer/app.js`

```js
// gui/renderer/app.js — App controller + UI State Adapter

const api = window.deepseek;
let currentLayer = "surface";
const eventBuffer = [];

// --- UI State Adapter ---

function adaptEvent(event) {
  eventBuffer.push(event);
  if (eventBuffer.length > 100) eventBuffer.shift();
  updateContextLayer();
  updateStatusBar(event);
}

function updateStatusBar(event) {
  if (event?.state?.entered) {
    document.getElementById("status-channel").textContent = event.state.entered;
  }
  // Poll usage stats every few seconds
}

function updateContextLayer() {
  const recent = eventBuffer.slice(-8);
  const log = document.getElementById("activity-log");
  log.innerHTML = recent.map(e => {
    const icon = { "user:message": "💬", "orchestrator:state": "🔄", "tool:call": "🔧", "tool:result": "✓", "permission:decision": "🔐" }[e.type] || "•";
    return `<div class="log-entry"><span class="log-icon">${icon}</span><span class="log-summary">${summarizeEvent(e)}</span></div>`;
  }).join("");
}

function summarizeEvent(e) {
  if (e.type === "user:message") return e.content?.slice(0, 60) || "";
  if (e.type === "orchestrator:state") return `${e.state?.exited} → ${e.state?.entered}`;
  if (e.type === "tool:call") return e.tool || "";
  return e.type;
}

// --- Three-layer control ---

function showLayer(layer) {
  document.getElementById("surface-layer").classList.toggle("hidden", layer !== "surface");
  document.getElementById("context-layer").classList.toggle("hidden", layer === "surface");
  currentLayer = layer;
}

// --- Message handling ---

async function sendMessage() {
  const input = document.getElementById("msg-input");
  const message = input.value.trim();
  if (!message) return;
  input.value = "";

  addMessage("user", message);
  showLayer("context"); // auto-expand context when agent works

  try {
    const response = await api.send(message, {});
    if (response?.result?.status === "awaiting_approval") {
      showApprovalBox(response.result.approval);
    } else if (response?.result) {
      addMessage("assistant", response.result.content || formatResult(response.result));
    } else if (response?.error) {
      addMessage("assistant", `❌ ${response.error}`);
    }
    showLayer("surface");
  } catch (err) {
    addMessage("assistant", `❌ ${err.message}`);
    showLayer("surface");
  }
}

function addMessage(role, content) {
  const div = document.createElement("div");
  div.className = `message ${role}`;
  div.textContent = content;
  document.getElementById("messages").appendChild(div);
  div.scrollIntoView({ behavior: "smooth" });
}

function formatResult(result) {
  return `状态: ${result.status || result.state || "complete"}`;
}

// --- Approval handling ---

function showApprovalBox(approval) {
  const box = document.getElementById("approval-box");
  box.className = "visible";
  box.innerHTML = `
    <div class="approval-card">
      <div class="approval-title">⚠ 需要确认</div>
      <div class="approval-type">${approval.type || "plan"}</div>
      <div class="approval-actions">
        <button id="btn-approve">✓ 同意</button>
        <button id="btn-deny">✗ 拒绝</button>
      </div>
    </div>
  `;
  document.getElementById("btn-approve").onclick = () => {
    api.approve("approval", "allow");
    box.className = "hidden";
  };
  document.getElementById("btn-deny").onclick = () => {
    api.approve("approval", "deny");
    box.className = "hidden";
  };
}

// --- Event listeners ---

document.getElementById("btn-send").onclick = sendMessage;
document.getElementById("msg-input").onkeydown = (e) => {
  if (e.key === "Enter") sendMessage();
};
document.getElementById("btn-context").onclick = () => {
  showLayer(currentLayer === "context" ? "surface" : "context");
};
document.getElementById("btn-close-context").onclick = () => showLayer("surface");

// --- Subscribe to kernel events ---

api.onKernelEvent((event) => {
  adaptEvent(event);
});

// --- Periodic status poll ---

setInterval(async () => {
  const usage = await api.getUsage();
  const state = await api.getState();
  const total = (usage.total_prompt_tokens || 0) + (usage.total_completion_tokens || 0);
  document.getElementById("status-tokens").textContent = total >= 1000 ? `${(total/1000).toFixed(1)}K tokens` : `${total} tokens`;
  document.getElementById("status-autonomy").textContent = state.autonomy || "gated";
  document.getElementById("status-channel").textContent = state.channel || "—";
  if (usage.requests > 0) {
    const denom = (usage.cache_hit_tokens || 0) + (usage.cache_miss_tokens || 0);
    const rate = denom > 0 ? Math.round(usage.cache_hit_tokens / denom * 100) : 0;
    document.getElementById("status-cache").textContent = `cache ${rate}%`;
  }
}, 2000);
```

### Pre-Execution Notes

1. **Electron uses CJS**: The main process and preload use `require()`. The kernel is imported via dynamic `import()` with `pathToFileURL` for Windows compatibility:
```js
const { pathToFileURL } = require("url");
const kernelPath = path.join(__dirname, "..", "src", "kernel", "kernel-api.js");
const { createKernel } = await import(pathToFileURL(kernelPath).href);
```

2. **Renderer is vanilla JS**: No React build step. Three layers via CSS toggling.

3. **CSP header**: `default-src 'self'; style-src 'self' 'unsafe-inline'; script-src 'self'`

4. **agent:send is fire-and-forget**: The IPC handler starts the task and returns `{ ok: true }`. Approval UI is driven by `orchestrator:state` events pushed from the kernel, NOT by awaiting `agent.send()`. This prevents the IPC promise from hanging on AwaitApproval.

5. **Default projectRoot**: Use `path.resolve(__dirname, "..")` (repo root), not `process.cwd()`. User can override with `--project=`.

6. **XSS prevention**: User content goes into `createElement` + `textContent`, never `innerHTML`.

7. **IPC handler guard**: Check `ipcMain.eventNames().includes(channel)` before registering.

8. **IPC whitelist**: 8 handlers — `agent:send`, `agent:approve`, `agent:interrupt`, `session:timeline`, `context:snapshot`, `model:usage`, `config:get`, `orchestrator:state`.

9. **GUI syntax check**: All three GUI JS files should be checked: `node --check gui/main.js gui/preload.js gui/renderer/app.js`
