重写 `src/tools/builtin/grep.js` 核心结构如下（保持 walk/read 与 max_matches）:

```js
import { readWorkspaceTextFile, walkWorkspaceFiles } from "../../workspace/path-safety.js";

export function createGrepTool({
  now = Date.now,
  totalTimeoutMs = 10_000,
  perFileTimeoutMs = 2_000
} = {}) {
  return {
    name: "grep",
    // ... 其它字段不变，version 可改为 "2.1"
    execute: async (params, context) => {
      const regex = new RegExp(params.pattern);
      const files = await walkWorkspaceFiles(context.projectRoot, params.path);
      const matches = [];
      const totalStart = now();
      let filesSearched = 0;
      let filesSkippedTimeout = 0;
      let timedOutScope = null;

      for (const filePath of files) {
        if (matches.length >= params.max_matches) break;
        if (now() - totalStart >= totalTimeoutMs) {
          timedOutScope = "total";
          break;
        }
        let file;
        try {
          file = await readWorkspaceTextFile(context.projectRoot, filePath, { maxBytes: 1024 * 1024 });
        } catch {
          continue;
        }
        filesSearched += 1;
        const fileStart = now();
        const lines = file.content.split(/\r?\n/);
        let fileTimedOut = false;
        for (let index = 0; index < lines.length; index++) {
          if (index % 64 === 0 || index === 0) {
            if (now() - totalStart >= totalTimeoutMs) {
              timedOutScope = "total";
              fileTimedOut = true;
              break;
            }
            if (now() - fileStart >= perFileTimeoutMs) {
              if (timedOutScope !== "total") timedOutScope = timedOutScope || "file";
              filesSkippedTimeout += 1;
              fileTimedOut = true;
              break;
            }
          }
          if (regex.test(lines[index])) {
            matches.push(`${filePath}:${index + 1}:${lines[index]}`);
            if (matches.length >= params.max_matches) break;
          }
        }
        if (timedOutScope === "total") break;
        // fileTimedOut: 已计入 skip，继续下一文件
      }

      const timed_out = timedOutScope != null || filesSkippedTimeout > 0;
      return {
        content: [{ type: "text", text: matches.join("\n") }],
        metadata: {
          matches: matches.length,
          timed_out,
          timed_out_scope: timedOutScope,
          files_skipped_timeout: filesSkippedTimeout,
          files_searched: filesSearched
        }
      };
    }
  };
}
```

注意：`timed_out` 与 scope 语义对齐设计——仅 file skip 时 `timed_out_scope` 为 `"file"`；total 优先 `"total"`。假时钟测试若 flaky，改为「每行都 `now()`」并稳定 n 序列。

```js
const CHILD_ENV_ALLOW = new Set([
  "path", "pathext", "systemroot", "comspec", "windir",
  "home", "userprofile", "homedrive", "homepath",
  "tmp", "temp", "tmpdir",
  "lang", "lc_all", "lc_ctype", "language"
].map((s) => s.toLowerCase()));

export function buildChildEnv(baseEnv = process.env, { allowExtra = [] } = {}) {
  const out = {};
  const lowerToActual = new Map();
  for (const key of Object.keys(baseEnv || {})) {
    lowerToActual.set(key.toLowerCase(), key);
  }
  for (const allowed of CHILD_ENV_ALLOW) {
    const actual = lowerToActual.get(allowed);
    if (actual != null && baseEnv[actual] != null) out[actual] = baseEnv[actual];
  }
  for (const extra of allowExtra) {
    if (typeof extra === "string" && extra in (baseEnv || {})) out[extra] = baseEnv[extra];
  }
  return out;
}

export function runProcess(argv, { cwd, timeoutMs = 30000, env } = {}) {
  // ...
  child = spawn(argv[0], argv.slice(1), {
    cwd,
    shell: false,
    windowsHide: true,
    env: env ?? buildChildEnv(process.env)
  });
  // rest unchanged
}
```

export const DEFAULT_MODELS = {

  think: "deepseek-v4-pro",

function channelsFor(models) {
  const m = { ...DEFAULT_MODELS, ...models };
  return {
    reply: { purpose: "reply", channel: "act", model: m.act, thinking: { type: "disabled" }, temperature: 0.2, max_tokens: 4096, stream: true },
    act: { purpose: "act", channel: "act", model: m.act, thinking: { type: "disabled" }, temperature: 0.1, max_tokens: 4096, stream: true },
    plan: { purpose: "plan", channel: "think", model: m.think, thinking: { type: "enabled" }, reasoning_effort: "high", temperature: 0.2, max_tokens: 8192, stream: false },
    review: { purpose: "review", channel: "think", model: m.think, thinking: { type: "enabled" }, reasoning_effort: "high", temperature: 0.2, max_tokens: 8192, stream: false },
    repair: { purpose: "repair", channel: "think", model: m.think, thinking: { type: "enabled" }, reasoning_effort: "high", temperature: 0.1, max_tokens: 8192, stream: false },
    fim: { purpose: "fim", channel: "fim", model: m.fim, max_tokens: 512 }
  };
}

export function routeModel({ purpose = "reply", complexity = "normal", explicitModel = null, models } = {}) {
  const CHANNELS = channelsFor(models);
  const key = purpose === "reply" && complexity === "high" ? "plan" : purpose;
  const profile = CHANNELS[key];
  if (!profile) throw new Error(`unknown DeepSeek purpose: ${purpose}`);
  return removeUndefined({ ...profile, purpose, model: explicitModel || profile.model });
}
